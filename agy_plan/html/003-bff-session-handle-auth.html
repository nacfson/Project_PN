+func (h *authHandler) session(w http.ResponseWriter, r *http.Request) {
+	r.Body = http.MaxBytesReader(w, r.Body, 4<<10)
+	var body struct {
+		Code string `json:"code"`
+	}
+	if err := json.NewDecoder(r.Body).Decode(&body); err != nil || len(strings.TrimSpace(body.Code)) != 32 {
+		writeError(w, http.StatusBadRequest, "invalid pickup code")
+		return
+	}
+	entry, ok := h.consumePickupCode(strings.TrimSpace(body.Code))

+		RefreshToken: entry.SessionHandle,

4. Security Audit & Automated Verification Steps

  • CWE-770 (Bounded Body): POST /api/auth/session wraps r.Body with http.MaxBytesReader(w, r.Body, 4<<10) and requires len(code) == 32.
  • CWE-352 (Login CSRF Binding): GET /api/auth/login sets __Host-pa_nonce and binds pa_bind SHA-256 base64url digest; GET /api/auth/callback verifies using subtle.ConstantTimeCompare.
  • CWE-384 (Single-Use 30s Pickup Code): Atomic lookup + deletion + TTL eviction under sync.Mutex.
  • Gitleaks Compliance: All test JWTs constructed dynamically via base64.RawURLEncoding.EncodeToString / jwx.
