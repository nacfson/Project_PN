5.2 Backend JWT Middleware (`backend/internal/http/auth_middleware.go`)

// Local in-cluster JWT verification replacing Central Auth HTTP validation
func authMiddleware(verifier *auth.JWTVerifier, ensurer platformUserEnsurer, logger admin.Logger) func(http.Handler) http.Handler {
    return func(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
            token := bearerToken(r)
            if token == "" {
                writeCodedError(w, http.StatusUnauthorized, "AUTH-101", "unauthorized", false)
                return
            }
            claims, err := verifier.Verify(r.Context(), token)
            if err != nil {
                writeCodedError(w, http.StatusUnauthorized, "AUTH-101", "unauthorized", false)
                return
            }
            authResult, err := ensurer.EnsurePlatformUser(r.Context(), claims.Issuer, claims.Subject, claims.Email, "google")
            if err != nil {
                writeCodedError(w, http.StatusInternalServerError, "AUTH-599", "internal error", true)
                return
            }
            next.ServeHTTP(w, r.WithContext(withUser(r.Context(), authResult.User)))
        })
    }
}
